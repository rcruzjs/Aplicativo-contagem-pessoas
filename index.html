CREATE TABLE fluxo.eventos (
  event_id STRING, timestamp TIMESTAMP, local STRING,
  direcao STRING, entradas_total INT64, saidas_total INT64,
  lotacao INT64, limite INT64, track_id INT64, origem STRING
) PARTITION BY DATE(timestamp);
